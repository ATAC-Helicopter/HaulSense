cmake_minimum_required(VERSION 3.20)
project(HaulSense LANGUAGES CXX)
add_subdirectory(ats-dualsense/daemon)
include(CTest)
if(BUILD_TESTING)
  add_executable(effects-tests tests/effects.cpp ats-dualsense/daemon/src/dualsense.cpp ats-dualsense/daemon/src/player_leds.cpp ats-dualsense/daemon/src/dashboard.cpp)
  target_compile_features(effects-tests PRIVATE cxx_std_20)
  target_compile_options(effects-tests PRIVATE -UNDEBUG)
  target_include_directories(effects-tests PRIVATE "${CMAKE_BINARY_DIR}/ats-dualsense/daemon")
  add_test(NAME effects COMMAND effects-tests)
  add_executable(plugin-tests tests/plugin.cpp)
  target_compile_features(plugin-tests PRIVATE cxx_std_20)
  target_compile_options(plugin-tests PRIVATE -fno-builtin -UNDEBUG)
  add_test(NAME plugin COMMAND plugin-tests)
  add_executable(dualsense-transport-tests tests/dualsense_transport.cpp ats-dualsense/daemon/src/dualsense.cpp)
  target_compile_features(dualsense-transport-tests PRIVATE cxx_std_20)
  target_compile_options(dualsense-transport-tests PRIVATE -UNDEBUG)
  add_test(NAME dualsense-transport COMMAND dualsense-transport-tests)
  add_executable(telemetry-fixture tests/telemetry_fixture.cpp)
  target_compile_features(telemetry-fixture PRIVATE cxx_std_20)
endif()
