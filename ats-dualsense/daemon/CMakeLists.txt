cmake_minimum_required(VERSION 3.20)
project(HaulSense VERSION 0.8.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../ui/index.html")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../ui/hud.html")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../ui/index.html" DASHBOARD_HTML)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../ui/hud.html" HUD_HTML)
configure_file(src/dashboard_asset.hpp.in dashboard_asset.hpp @ONLY)
configure_file(src/hud_asset.hpp.in hud_asset.hpp @ONLY)
add_executable(haulsense src/main.cpp src/dualsense.cpp src/player_leds.cpp src/dashboard.cpp)
target_include_directories(haulsense PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(haulsense PRIVATE -Wall -Wextra -Wpedantic)
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_supported)
if(ipo_supported)
  set_property(TARGET haulsense PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_options(haulsense PRIVATE -Os -ffunction-sections -fdata-sections)
  target_link_options(haulsense PRIVATE -Wl,--gc-sections -Wl,--as-needed -s)
endif()
install(TARGETS haulsense RUNTIME DESTINATION bin)
