cmake_minimum_required(VERSION 3.20)
project(HaulSense VERSION 1.0.0 LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../VERSION")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../VERSION" HAULSENSE_VERSION)
string(STRIP "${HAULSENSE_VERSION}" HAULSENSE_VERSION)
configure_file(src/version.hpp.in version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../ui/index.html")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../ui/hud.html")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../ui/navigation.js")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../ui/navigation.js" NAVIGATION_JS)
configure_file(src/navigation_asset.hpp.in navigation_asset.hpp @ONLY)
foreach(asset scene routing dashboard hud persistence jobs)
  if(asset STREQUAL "scene")
    set(asset_path "${CMAKE_CURRENT_SOURCE_DIR}/../ui/scene.bundle.js")
  elseif(asset STREQUAL "routing")
    set(asset_path "${CMAKE_CURRENT_SOURCE_DIR}/../ui/routing-worker.js")
  else()
    set(asset_path "${CMAKE_CURRENT_SOURCE_DIR}/../ui/${asset}.js")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${asset_path}")
  file(READ "${asset_path}" ASSET_JS)
  set(ASSET_NAME "${asset}_js")
  if(asset STREQUAL "dashboard" OR asset STREQUAL "hud")
    configure_file(src/js_asset.hpp.in "${asset}_js_asset.hpp" @ONLY)
  else()
    configure_file(src/js_asset.hpp.in "${asset}_asset.hpp" @ONLY)
  endif()
endforeach()
find_package(Threads REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../ui/index.html" DASHBOARD_HTML)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../ui/hud.html" HUD_HTML)
configure_file(src/dashboard_asset.hpp.in dashboard_asset.hpp @ONLY)
configure_file(src/hud_asset.hpp.in hud_asset.hpp @ONLY)
add_executable(haulsense src/main.cpp src/dualsense.cpp src/player_leds.cpp src/dashboard.cpp src/route_reader.cpp src/semaphore_reader.cpp src/job_journal.cpp)
target_link_libraries(haulsense PRIVATE Threads::Threads)
target_include_directories(haulsense PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(haulsense PRIVATE -Wall -Wextra -Wpedantic)
include(CheckIPOSupported)
check_ipo_supported(RESULT ipo_supported)
if(ipo_supported)
  set_property(TARGET haulsense PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_options(haulsense PRIVATE -Os -ffunction-sections -fdata-sections)
  target_link_options(haulsense PRIVATE -Wl,--gc-sections -Wl,--as-needed -s)
endif()
install(TARGETS haulsense RUNTIME DESTINATION bin)
